/* Печать значения: десятичное + двоичное представление в нужном числе байтов */
void pr(const char *name, long long dec, int bytes)
{
    unsigned int bits = (unsigned int)dec;   /* младшие байты = машинное представление */
    printf("  %-18s = %12lld   ", name, dec);
    for (int b = bytes * 8 - 1; b >= 0; b--)
    {
        printf("%d", (bits >> b) & 1);
        if (b % 8 == 0 && b != 0) printf(" ");
    }
    printf("   (%d байт)\n", bytes);
}

/* Четыре «цепочки» преобразований из задания 3 */
void chains(char c, unsigned char uc, int i, unsigned int ui)
{
    short s;
    unsigned short us;
    char c2;
    unsigned char uc2;
    int i2;
    unsigned int ui2;

    printf(" Цепочка char -> short int -> int\n");
    s = c;   i2 = s;
    pr("char", c, 1);  pr("short int", s, 2);  pr("int", i2, 4);

    printf(" Цепочка unsigned char -> unsigned short int -> unsigned int\n");
    us = uc; ui2 = us;
    pr("unsigned char", uc, 1);  pr("unsigned short int", us, 2);  pr("unsigned int", ui2, 4);

    printf(" Цепочка int -> short int -> char\n");
    s = i;   c2 = s;
    pr("int", i, 4);  pr("short int", s, 2);  pr("char", c2, 1);

    printf(" Цепочка unsigned int -> unsigned short int -> unsigned char\n");
    us = ui; uc2 = us;
    pr("unsigned int", ui, 4);  pr("unsigned short int", us, 2);  pr("unsigned char", uc2, 1);
}
